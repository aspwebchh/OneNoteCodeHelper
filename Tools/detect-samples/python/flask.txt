from flask import Flask, jsonify, request

app = Flask(__name__)
users = {}


@app.route("/users/<int:uid>", methods=["GET"])
def get_user(uid):
    user = users.get(uid)
    if user is None:
        return jsonify({"error": "not found"}), 404
    return jsonify(user)


@app.post("/users")
def create_user():
    data = request.get_json()
    uid = len(users) + 1
    users[uid] = {"id": uid, "name": data["name"]}
    return jsonify(users[uid]), 201
